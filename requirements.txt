#   pip install -e '.[torch]'
#   pip install -e '.[jax,torch]'
#
# Note the JAX extra pins numpy<2.5: the pristine reference implementation under reference/
# relies on numpy accepting nested np.random.SeedSequence input, which numpy 2.5 removed.
